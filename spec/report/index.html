// specs/engine.qnt — Spec 4: the topology engine (engine/src/dizzy_topology_engine),
// over an ARBITRARY feature graph and an arbitrary rule-3 placement.
//
// The topology engine is the Simple Engine (ext/dizzy/docs/design/simple-engine.md)
// with lanes: an inline lane served on the host thread, and a pool lane whose
// procedures run in worker processes while the host thread keeps claiming. Unlike
// st.qnt (the pre-Simple-Engine `dizzy.engine.st`), the store's transaction
// boundaries are what make it correct, so they are modelled one step each:
//
//   claim        attempt `running`, durably, before anything runs  (_claim_next)
//   append+ack   the block AND the command's `done`, ONE transaction (_unit, _commit)
//   settleFold   fold one event, then its policies enqueue by derived id with
//                INSERT OR IGNORE                                  (Engine._settle)
//   advance      the cursor moves past that event — its own transaction
//
// and a CRASH may fall between any two of them. The store (log, command table,
// cursor) survives; process memory (read model, in-flight pool work, the unit the
// host thread is running) does not. Reopening refolds the read model up to the
// cursor (spec §4), settles what lies past it, and re-runs every attempt still
// `running` under its own id.
//
// What the host thread serialises, and why claims wait for `settled`: after an
// append the engine settles the new events before it does anything else, and a
// reopen settles before its first claim. So a claim always sees cursor == log
// length — zero lag at the moment a procedure reads.
//
// Placement: `pooled` ⊆ PROCS is drawn at init. Topology rule 3 says only a
// procedure that declares no `queries:` may be pooled; `rule3` is drawn too, so the
// suite can show the rule is NECESSARY (witnessStalePool) as well as sufficient
// (freshReads). A pool child never sees the host's read model; when rule 3 is broken
// it is modelled as reading the snapshot taken when its job was dispatched — the most
// charitable thing an implementation could do — and that is still stale.
//
// The headline results, against st.qnt's:
//   • effectsOnce holds UNCONDITIONALLY. st needed the application to guard, and
//     only guardable procedures could (witnessDoubleAppend there). Here the append
//     and the ack are one transaction, so a replay can never append a second block.
//   • freshReads holds with pools (given rule 3), although the log's ORDER is no
//     longer the claim order (witnessOutOfOrder).
//   • freshDispatch and dispatchComplete survive crashes, including a crash
//     between a policy's enqueue and the cursor (witnessResettle): the re-settle
//     re-derives the same ids and the store ignores them.
//
//   quint run    engine.qnt --invariant=allInvariants --max-samples=3000 --max-steps=25
//   quint run    engine.qnt --invariant='not(witnessOutOfOrder)'   # expect violation = reachable
//   quint verify engine.qnt --invariant=allInvariants --max-steps=4

module engine {
  import graph.* from "./graph"

  pure val MAX_SUBMISSIONS = 2   // host-originated commands
  pure val MAX_LOG = 6           // a policy cycle can be genuinely infinite; bound it
  pure val MAX_CRASHES = 2
  pure val MAX_FAILS = 1         // element raises (or a pool worker dies)
  pure val POOL_WORKERS = 2      // one pool lane, this wide

  type Row = { cmd: Command, status: str }        // status: queued | running | done | failed
  type Job = { idx: int, view: View }             // a pool job: table row + what it could read

  // --- State: the store (durable) -----------------------------------------
  var log: List[Event]           // events table
  var table: List[Row]           // commands table, in enqueue (seq) order
  var cursor: int                // cursors table: events settled through here
  // --- State: the host process (volatile) --------------------------------
  var models: View               // the read model
  var folded: int                // ghost: models == fold of log[0, folded)
  var current: List[int]         // the inline unit the host thread is running (0 or 1 row)
  var inflight: Set[Job]         // pool jobs dispatched and not yet committed
  // --- Configuration (fixed after init) ----------------------------------
  var g: Graph
  var pooled: Set[str]           // procedures placed on the pool lane
  var rule3: bool                // does the placement obey topology rule 3?
  // --- Ghosts ------------------------------------------------------------
  var submitted: int
  var crashes: int
  var fails: int
  var reran: int                 // claims of a row already `running` (recovery)
  var ignored: int               // policy enqueues the store ignored (re-settle)

  pure def isReaderProc(gr: Graph, p: str): bool = gr.reads.get(p) != Set()

  action init = {
    nondet reads = READS.oneOf()
    nondet emits = EMITS.oneOf()
    nondet folds = FOLDS.oneOf()
    nondet reacts = REACTS.oneOf()
    nondet dispatches = DISPATCHES.oneOf()
    nondet interactive = INTERACTIVE.oneOf()
    nondet pl = PROCS.powerset().oneOf()
    val gr = { reads: reads, emits: emits, folds: folds, reacts: reacts, dispatches: dispatches, interactive: interactive }
    all {
      g' = gr,
      pooled' = pl,
      rule3' = pl.forall(p => not(isReaderProc(gr, p))),
      log' = [], table' = [], cursor' = 0,
      models' = emptyView, folded' = 0, current' = [], inflight' = Set(),
      submitted' = 0, crashes' = 0, fails' = 0, reran' = 0, ignored' = 0,
    }
  }

  // --- Helpers -------------------------------------------------------------
  val settled = cursor == log.length() and folded == cursor
  val busyIdx = inflight.map(j => j.idx)
  val poolRoom = inflight.size() < POOL_WORKERS

  pure def ids(t: List[Row]): Set[CmdId] = t.indices().map(i => t[i].cmd.id)
  pure def setStatus(t: List[Row], i: int, s: str): List[Row] =
    t.replaceAt(i, { cmd: t[i].cmd, status: s })
  pure def minOf(s: Set[int]): int = s.fold(-1, (a, i) => if (a == -1 or i < a) i else a)

  // _claim_next: an interrupted attempt first (in seq order, its lane must have room),
  // then interactive before batch, then seq — among rows whose lane has room.
  def hasRoom(i: int): bool = not(pooled.contains(table[i].cmd.proc)) or poolRoom
  def claimIndex: int = {
    val recovering = table.indices().filter(i =>
      table[i].status == "running" and not(busyIdx.contains(i)) and hasRoom(i))
    val queued = table.indices().filter(i => table[i].status == "queued" and hasRoom(i))
    val urgent = queued.filter(i => g.interactive.contains(table[i].cmd.proc))
    if (recovering != Set()) minOf(recovering)
    else if (urgent != Set()) minOf(urgent)
    else minOf(queued)
  }

  // Append a block and ack its row: ONE transaction.
  action appendAndAck(i: int, view: View): bool = {
    val emitted = runProcedure(g, table[i].cmd, view, log.length())
    all {
      log' = log.concat(emitted),
      table' = setStatus(table, i, "done"),
    }
  }

  action unchangedConfig = all { g' = g, pooled' = pooled, rule3' = rule3 }

  // --- Host submits a command ----------------------------------------------
  action submit = {
    nondet p = PROCS.oneOf()
    all {
      submitted < MAX_SUBMISSIONS,
      table' = table.append({ cmd: { id: { origin: "host", at: submitted, target: p }, proc: p, view: emptyView }, status: "queued" }),
      submitted' = submitted + 1,
      log' = log, cursor' = cursor, models' = models, folded' = folded, current' = current, inflight' = inflight,
      crashes' = crashes, fails' = fails, reran' = reran, ignored' = ignored, unchangedConfig,
    }
  }

  // --- Claim (host thread, between units, only when settled) ----------------
  action claim = all {
    current == [],
    settled,
    log.length() < MAX_LOG,
    val i = claimIndex
    all {
      i >= 0,
      table' = setStatus(table, i, "running"),
      reran' = if (table[i].status == "running") reran + 1 else reran,
      if (pooled.contains(table[i].cmd.proc))
        // A pool child gets no read model. Under rule 3 the procedure reads nothing, so
        // the snapshot is irrelevant; without it, this is the freshest view it could have.
        all { inflight' = inflight.union(Set({ idx: i, view: models })), current' = current }
      else
        all { current' = [i], inflight' = inflight },
    },
    log' = log, cursor' = cursor, models' = models, folded' = folded,
    submitted' = submitted, crashes' = crashes, fails' = fails, ignored' = ignored, unchangedConfig,
  }

  // --- The inline unit: run against the live read model, append + ack ------
  action runInline = all {
    current.length() == 1,
    val i = current[0]
    appendAndAck(i, models),
    current' = [],
    cursor' = cursor, models' = models, folded' = folded, inflight' = inflight,
    submitted' = submitted, crashes' = crashes, fails' = fails, reran' = reran, ignored' = ignored, unchangedConfig,
  }

  // --- A pool job finishes; the host thread commits it (only when settled) ---
  action completePool = all {
    current == [],
    settled,
    inflight != Set(),
    nondet j = inflight.oneOf()
    all {
      appendAndAck(j.idx, j.view),
      inflight' = inflight.exclude(Set(j)),
    },
    current' = current, cursor' = cursor, models' = models, folded' = folded,
    submitted' = submitted, crashes' = crashes, fails' = fails, reran' = reran, ignored' = ignored, unchangedConfig,
  }

  // --- An element raises (or a pool worker dies): attempt failed, nothing appended
  action failInline = all {
    current.length() == 1,
    fails < MAX_FAILS,
    table' = setStatus(table, current[0], "failed"),
    current' = [], fails' = fails + 1,
    log' = log, cursor' = cursor, models' = models, folded' = folded, inflight' = inflight,
    submitted' = submitted, crashes' = crashes, reran' = reran, ignored' = ignored, unchangedConfig,
  }
  action failPool = all {
    current == [],
    settled,
    fails < MAX_FAILS,
    inflight != Set(),
    nondet j = inflight.oneOf()
    all {
      table' = setStatus(table, j.idx, "failed"),
      inflight' = inflight.exclude(Set(j)),
    },
    fails' = fails + 1,
    log' = log, cursor' = cursor, models' = models, folded' = folded, current' = current,
    submitted' = submitted, crashes' = crashes, reran' = reran, ignored' = ignored, unchangedConfig,
  }

  // --- The host retries a failed command: a new attempt, never automatic ----
  action retry = {
    nondet i = table.indices().oneOf()
    all {
      table.length() > 0,
      table[i].status == "failed",
      table' = setStatus(table, i, "queued"),
      log' = log, cursor' = cursor, models' = models, folded' = folded, current' = current, inflight' = inflight,
      submitted' = submitted, crashes' = crashes, fails' = fails, reran' = reran, ignored' = ignored, unchangedConfig,
    }
  }

  // --- Settle one event: fold, then its policies enqueue (INSERT OR IGNORE) --
  action settleFold = all {
    current == [],
    folded == cursor,
    cursor < log.length(),
    val e = log[cursor]
    val v2 = applyEvent(g, models, e, cursor)
    val dispatched = runPolicies(g, e, cursor, v2)
    val r = dispatched.foldl({ t: table, dup: 0 }, (acc, c) =>
      if (ids(acc.t).contains(c.id)) { t: acc.t, dup: acc.dup + 1 }
      else { t: acc.t.append({ cmd: c, status: "queued" }), dup: acc.dup })
    all {
      models' = v2,
      folded' = folded + 1,
      table' = r.t,
      ignored' = ignored + r.dup,
    },
    log' = log, cursor' = cursor, current' = current, inflight' = inflight,
    submitted' = submitted, crashes' = crashes, fails' = fails, reran' = reran, unchangedConfig,
  }

  // --- ... and only then the cursor moves past it ---------------------------
  action advance = all {
    folded == cursor + 1,
    cursor' = folded,
    log' = log, table' = table, models' = models, folded' = folded, current' = current, inflight' = inflight,
    submitted' = submitted, crashes' = crashes, fails' = fails, reran' = reran, ignored' = ignored, unchangedConfig,
  }

  // --- Crash and reopen: memory is gone, the store is not -------------------
  // _recover refolds a fresh read model to the cursor; settling past it is the
  // ordinary settle steps, which claims wait for.
  action crash = all {
    crashes < MAX_CRASHES,
    models' = foldPrefix(g, log, cursor),
    folded' = cursor,
    current' = [],
    inflight' = Set(),
    crashes' = crashes + 1,
    log' = log, table' = table, cursor' = cursor,
    submitted' = submitted, fails' = fails, reran' = reran, ignored' = ignored, unchangedConfig,
  }

  val idle = current == [] and inflight == Set() and settled and
    table.indices().forall(i => table[i].status == "done" or table[i].status == "failed")
  action quiescent = all {
    idle or log.length() >= MAX_LOG,
    submitted == MAX_SUBMISSIONS,
    log' = log, table' = table, cursor' = cursor, models' = models, folded' = folded,
    current' = current, inflight' = inflight,
    submitted' = submitted, crashes' = crashes, fails' = fails, reran' = reran, ignored' = ignored, unchangedConfig,
  }

  action step = any {
    submit, claim, runInline, completePool, failInline, failPool, retry,
    settleFold, advance, crash, quiescent,
  }

  // --- Invariants ----------------------------------------------------------

  // E1. The read model is a fold of the log (the refold-on-open guarantee).
  val modelsAreFold = models == foldPrefix(g, log, folded)

  // E2. The cursor never claims ground the read model has not covered.
  val cursorSound = cursor <= folded and folded <= cursor + 1 and folded <= log.length()

  // E3. Every block is a complete, contiguous, ordered run of its procedure.
  val blocksWellFormed = blocksWellFormedOf(g, log)

  // E4. EXACTLY-ONCE EFFECTS, unconditionally: every cause owns one block. No
  //     application guard, no `guardable` side-condition — the transaction does it.
  val effectsOnce = effectsOnceOf(log)

  // E5. Append and ack are one transaction: a block's command is `done`, and a
  //     `done` command that emits has its block.
  pure def statusOf(t: List[Row], id: CmdId): Set[str] =
    t.indices().filter(i => t[i].cmd.id == id).map(i => t[i].status)
  val ackAtomic = all {
    log.indices().forall(i => statusOf(table, log[i].cause) == Set("done")),
    table.indices().forall(i => (table[i].status == "done" and g.emits.get(table[i].cmd.proc) != Set())
      implies log.indices().exists(k => log[k].cause == table[i].cmd.id)),
    // a failed or unrun attempt appended nothing
    table.indices().forall(i => table[i].status != "done"
      implies not(log.indices().exists(k => log[k].cause == table[i].cmd.id))),
  }

  // E6. Causality of ids.
  val causal = causalOf(g, log)

  // E7. Fresh reads — sequential consistency of every decision — given rule 3.
  val freshReads = rule3 implies freshReadsOf(g, log)

  // E8. Policies read exactly the fold through their trigger, across crashes, in the
  //     log and in the queue.
  val freshDispatch = freshDispatchOf(g, log)
  val freshQueue = {
    val pv = prefixViews(g, log)
    table.indices().forall(i => {
      val c = table[i].cmd
      c.id.origin != "host" implies c.view == restrict(g, c.id.origin, pv.get(c.id.at + 1))
    })
  }

  // E9. Nothing settled is lost: every reacting policy of every settled event has
  //     its command in the table.
  val dispatchComplete =
    log.indices().forall(pos => pos < cursor implies
      POLICIES.forall(y => g.reacts.get(y).contains(log[pos].kind) implies
        g.dispatches.get(y).forall(p => ids(table).contains({ origin: y, at: pos, target: p }))))

  // E10. One row per command id.
  val idsDistinct = ids(table).size() == table.length()

  // E11. Process memory points only at rows that are really running.
  val memorySound = all {
    current.indices().forall(k => table[current[k]].status == "running"),
    inflight.forall(j => table[j.idx].status == "running"),
    current.indices().forall(k => not(busyIdx.contains(current[k]))),
    inflight.size() <= POOL_WORKERS,
  }

  // E12. Nothing waits behind an empty engine: when idle, every command finished.
  val noLostWork = (idle and log.length() < MAX_LOG) implies
    table.indices().forall(i => table[i].status == "done" or table[i].status == "failed")

  val allInvariants = all {
    modelsAreFold, cursorSound, blocksWellFormed, effectsOnce, ackAtomic, causal,
    freshReads, freshDispatch, freshQueue, dispatchComplete, idsDistinct, memorySound, noLostWork,
  }

  // --- Witnesses (should be REACHABLE: `--invariant=not(w)` must violate) ---
  val witnessCascade    = log.indices().exists(i => log[i].cause.origin != "host")
  val witnessReader     = log.indices().exists(i => log[i].view != emptyView)
  val witnessPooledRun  = log.indices().exists(i => pooled.contains(log[i].cause.target))
  // Two pool jobs genuinely in flight at once.
  val witnessConcurrent = inflight.size() == POOL_WORKERS
  // The log is not in claim order: a block lands after a block whose command was
  // enqueued LATER. Pools make the order a race; the reads stay fresh regardless.
  pure def seqOf(t: List[Row], id: CmdId): int =
    t.indices().filter(i => t[i].cmd.id == id).fold(-1, (a, i) => i)
  val witnessOutOfOrder = rule3 and log.indices().exists(i => log.indices().exists(k =>
    i < k and seqOf(table, log[i].cause) > seqOf(table, log[k].cause)))
  val witnessPriority   = table.length() >= 2 and settled and current == [] and
    claimIndex > 0 and table.indices().forall(i => table[i].status != "running")
  val witnessRecovered  = reran > 0                  // an interrupted attempt re-ran under its id
  val witnessCrashAfterAppend = crashes > 0 and log.length() > 0
  val witnessResettle   = ignored > 0                // a re-settle re-derived ids the store held
  val witnessFailed     = table.indices().exists(i => table[i].status == "failed")
  // Every failure was retried by the host and the run still finished clean.
  val witnessRetried    = fails > 0 and idle and table.indices().forall(i => table[i].status == "done")
  val witnessBound      = log.length() >= MAX_LOG
  // Rule 3 is NECESSARY: pool a reader and it decides on a stale model.
  val witnessStalePool  = not(rule3) and not(freshReadsOf(g, log))
}

minor Crash between append and fold, and rebuild/refold, are covered only implicitly; the spec should say that I2 *is* the rebuild guarantee and that its positions are the local watermark

In the code, append (loop.py:333) precedes fold (:345); a crash between them leaves the log one entry ahead of the models with the row still `running`. On restart models.db is disposable (command_queue.py:10-11) and `rebuild` refolds `store.iterate()` (rebuild.py:54), then the row replays (finding 1). The spec has no such window — `zeroLag` and `modelsAreFold` are invariants — and that is defensible for st because `modelsAreFold` is exactly what rebuild recomputes: models == foldPrefix(g, log, |log|). But nothing says so, and the spec's `cursor`/positions are silently the local insertion watermark that topology.md:361-366 distinguishes from canonical order; rebuild iterates canonical order, which coincides with insertion order only while no replicated fact has entered the store.

// specs/st.qnt — Spec 1: the single-threaded engine (dizzy.engine.st), over an
// ARBITRARY feature graph.
//
// One hybrid consumer group: one inline lane over the durable command queue,
// one cursor over the log, both served by one thread. Engine.run_command —
// claim → run_procedure → ingest(append → fold → commit → run policies →
// dispatch) — is ONE atomic step (engine/loop.py:254-402). That atomicity is
// what "sequentially consistent" means for Dizzy; lag is zero by construction
// (`cursor == log.length()` is an invariant here and a *choice* in groups.qnt).
//
// Nondeterminism, deliberately:
//   • the feature graph (init)                — every feature of the bounded size
//   • `guarded` (init)                        — whether the application bothers to
//                                                dedupe redeliveries. The ENGINE does
//                                                not (st/executor.py:49-56: a replay
//                                                "appends the fact again ... nothing
//                                                dedups it"); topology.md:347-351 asks
//                                                it of the application. The guard is
//                                                modelled as the only one a procedure
//                                                CAN implement — "my own block is
//                                                already in a model I read" — so
//                                                exactly-once holds only for
//                                                `guardable` procedures (graph.qnt).
//   • which command the host submits, and when
//   • redelivery: a crash between ingest and ack re-queues the claimed row at
//     the FRONT of its priority level (command_queue.py:409-414, claim order
//     `priority DESC, id`)
//
// Out of scope here, by design (see groups.qnt / Spec 3 seeds): `executor: pool`
// lanes, an element raising mid-emission (a partial block), replication
// (`add_replicated`, which breaks prefix positions), host `put` interleaving
// between two dispatches of one cascade.
//
//   quint run    st.qnt --invariant=allInvariants --max-samples=3000 --max-steps=20
//   quint run    st.qnt --invariant='not(witnessDoubleAppend)'   # expect violation = reachable
//   quint verify st.qnt --invariant=allInvariants --max-steps=3  # Apalache; deeper is infeasible today

module st {
  import graph.* from "./graph"

  pure val MAX_SUBMISSIONS = 2   // host-originated commands
  pure val MAX_LOG = 6           // a policy cycle can be genuinely infinite; bound it

  // --- State -------------------------------------------------------------
  var g: Graph               // the feature (fixed after init)
  var guarded: bool          // application-level dedupe assumption (fixed after init)
  var queue: List[Command]   // st.DurableCommandQueue — one lane
  var log: List[Event]       // EventStore, the only shared truth
  var cursor: int            // folded prefix length
  var models: View           // read models (positions folded per model)
  var ran: Set[Command]      // commands whose procedure has run (a redelivery replays the row verbatim)
  var submitted: int
  var noops: int             // ghost: redeliveries the application guard absorbed

  action init = {
    nondet reads = READS.oneOf()
    nondet emits = EMITS.oneOf()
    nondet folds = FOLDS.oneOf()
    nondet reacts = REACTS.oneOf()
    nondet dispatches = DISPATCHES.oneOf()
    nondet interactive = INTERACTIVE.oneOf()
    nondet gd = Set(true, false).oneOf()
    all {
      g' = { reads: reads, emits: emits, folds: folds, reacts: reacts, dispatches: dispatches, interactive: interactive },
      guarded' = gd,
      queue' = [], log' = [], cursor' = 0, models' = emptyView, ran' = Set(), submitted' = 0, noops' = 0,
    }
  }

  // --- Host submits a command ------------------------------------------------
  action submit = {
    nondet p = PROCS.oneOf()
    all {
      submitted < MAX_SUBMISSIONS,
      queue' = queue.append({ id: { origin: "host", at: submitted, target: p }, proc: p, view: emptyView }),
      submitted' = submitted + 1,
      g' = g, guarded' = guarded, log' = log, cursor' = cursor, models' = models, ran' = ran, noops' = noops,
    }
  }

  // --- Claim: LaneExecutor's two-level priority (command_queue.py:248-251) ---
  // The earliest interactive command wins; otherwise the earliest command.
  pure def claimIndex(gr: Graph, q: List[Command]): int = {
    val urgent = q.indices().filter(i => gr.interactive.contains(q[i].proc))
    if (urgent != Set()) urgent.fold(-1, (a, i) => if (a == -1 or i < a) i else a)
    else 0
  }
  pure def without(q: List[Command], i: int): List[Command] =
    q.slice(0, i).concat(q.slice(i + 1, q.length()))

  // --- ingest: append, fold, commit, run policies — per event, in order ------
  // Mirrors Engine.apply_event exactly; the policies' view is the models
  // as folded *through this event* (fold → commit → policies).
  pure def ingest(gr: Graph, l: List[Event], v: View, events: List[Event]):
      { log: List[Event], models: View, dispatched: List[Command] } =
    events.foldl({ log: l, models: v, dispatched: [] }, (acc, e) => {
      val pos = acc.log.length()
      val v2 = applyEvent(gr, acc.models, e, pos)
      {
        log: acc.log.append(e),
        models: v2,
        dispatched: acc.dispatched.concat(runPolicies(gr, e, pos, v2)),
      }
    })

  // --- The engine step -----------------------------------------------------
  action runCommand = all {
    queue.length() > 0,
    log.length() < MAX_LOG,
    val i = claimIndex(g, queue)
    val cmd = queue[i]
    val rest = without(queue, i)
    if (guarded and seesOwnBlock(g, log, models, cmd))
      // application-level guard: the procedure sees its own earlier block in a model it reads
      all { queue' = rest, log' = log, cursor' = cursor, models' = models, ran' = ran, noops' = noops + 1 }
    else {
      val emitted = runProcedure(g, cmd, models, log.length())
      val r = ingest(g, log, models, emitted)
      all {
        log' = r.log,
        models' = r.models,
        cursor' = r.log.length(),
        queue' = rest.concat(r.dispatched),  // dispatch is to the external queue (loop.py:152-158)
        ran' = ran.union(Set(cmd)),
        noops' = noops,
      }
    },
    g' = g, guarded' = guarded, submitted' = submitted,
  }

  // --- At-least-once: a crash between ingest and ack recovers the row --------
  // The row keeps its id, so it re-enters at the front of its priority level.
  action redeliver = {
    nondet c = ran.oneOf()
    all {
      ran != Set(),
      not(queue.indices().exists(i => queue[i].id == c.id)),   // one DB row: cannot be queued twice
      queue' = [c].concat(queue),
      g' = g, guarded' = guarded, log' = log, cursor' = cursor, models' = models, ran' = ran, submitted' = submitted, noops' = noops,
    }
  }

  // End of run: explicit so the checker does not call it a deadlock.
  action quiescent = all {
    queue.length() == 0 or log.length() >= MAX_LOG,
    submitted == MAX_SUBMISSIONS,
    g' = g, guarded' = guarded, queue' = queue, log' = log, cursor' = cursor, models' = models, ran' = ran, submitted' = submitted, noops' = noops,
  }

  action step = any { submit, runCommand, redeliver, quiescent }

  // --- Invariants ----------------------------------------------------------

  // I1. No lag.
  val zeroLag = cursor == log.length()

  // I2. Convergence: carried models are a refold of the log (this IS the rebuild guarantee).
  val modelsAreFold = models == foldPrefix(g, log, log.length())

  // I3. Every block is a complete run of its procedure, in emit order.
  val blocksWellFormed = blocksWellFormedOf(g, log)

  // I3'. Exactly-once effects — for GUARDABLE procedures, when the application
  //      guards. A procedure that reads no model folding its own emits cannot
  //      recognise its replay; nothing in the engine helps it (executor.py:49-56).
  val effectsOnce = guarded implies effectsOnceGuardableOf(g, log)

  // I4. Causality of ids.
  val causal = causalOf(g, log)

  // I5. Fresh reads — THE sequential-consistency property, per block.
  val freshReads = freshReadsOf(g, log)

  // I6. Policies read fresh too, visible at log level via dispatchView …
  val freshDispatch = freshDispatchOf(g, log)
  // … and in the queue, for commands not yet run.
  val freshQueue = {
    val pv = prefixViews(g, log)
    queue.indices().forall(i => {
      val c = queue[i]
      c.id.origin != "host" implies
        c.view == restrict(g, c.id.origin, pv.get(c.id.at + 1))
    })
  }

  // I7. Completeness — an engine that drops work must FAIL these:
  //   every reacting policy of every folded event has dispatched its command …
  val queuedIds = queue.indices().map(i => queue[i].id)
  val ranIds    = ran.map(c => c.id)

// specs/groups.qnt — Spec 2: consumer groups, i.e. the distributed engine.
//
// The same feature graph, split across GROUPS — each a process (or device)
// with ONE cursor over the shared log and ONE lane of the durable command
// queue (dizzy/engine/groups.py, topology.md "What a group worker does").
//
// Placement is a map `owner : PROCS ∪ POLICIES ∪ MODELS → GROUPS`, chosen
// nondeterministically at init alongside the graph, so a bounded check ranges
// over every (feature, topology) pair of the bounded size — including the
// one-group placement, which must behave exactly like st (`singleGroupIsSt`).
// Making the MODEL the unit of placement bakes in rule 1. Rule 3 (worker
// counts) is out of scope: every group is one lane.
//
// Shared vs per-group:
//   • log          shared, append-only, totally ordered (the EventStore)
//   • substrate    shared read models (one sqla DB); model m is folded ONLY by
//                  owner(m), so it reflects owner(m)'s progress, not the tail
//   • cursor[h]    per group, durable; `foldOne` advances it by one entry,
//                  running h's projections then h's policies on that entry
//   • lane[h]      per group; commands whose procedure h owns
//   • pending[h]   the command h has claimed and not yet run
//   • transport    a doorbell only, so not modelled: a fold may happen at any
//                  time ("a dropped notification costs latency, never work")
//
// The worker's cycle (groups.py:419-426) is
//     fold_to_tail(); claim(); fold_to_tail(); execute(); fold_to_tail()
// with NO lock between the middle fold returning and the procedure's first
// read (groups.py:44-50). So it is two steps here: `claim` (guarded by
// cursor-at-tail: the middle fold has returned) and `execute` (unguarded: other
// groups may have appended in between). That window is real, and it is why
// the theorem below is NOT "a rule-6-legal split reads what st reads":
//
//   what groups.py promises (:27-29): the read model has committed THROUGH
//   THE COMMAND'S CAUSE before the procedure runs         → rule6 ⇒ causeFolded
//   what st gives: the read model is folded to the tail   → freshReads
//
// `witnessClaimWindow` shows rule6 ∧ ¬freshReads is reachable: a rule-6-legal
// deployment can still decide on a model that is behind the log, i.e. the
// "byte-identical models" claim in topology.md:322-326 does not hold for the
// worker as written. Closing the window (a lock, or re-checking the tail after
// the claim) would make `freshReads` provable again — that is a design choice
// for Dizzy, surfaced here rather than assumed away.
//
// Faults modelled: at-least-once command redelivery (`redeliver`) and cursor
// loss (`rewind`: commit-then-save, cursors.py:33-46 — models keep their
// committed state, the cursor goes back, entries are re-folded and policies
// re-dispatch under the same deterministic ids).
//
// Deferred to Spec 3 (seeds): per-device substrate replicas, sharded
// event-only policy groups (k/N), a group folding its own appends ahead of
// earlier foreign entries, lane-file loss.
//
//   quint run    groups.qnt --invariant=allInvariants --max-samples=3000 --max-steps=20
//   quint run    groups.qnt --invariant='not(witnessClaimWindow)'  # expect violation = reachable
//   quint verify groups.qnt --invariant=allInvariants --max-steps=3

module groups {
  import graph.* from "./graph"

  pure val PLACEABLE = ELEMS.union(MODELS)
  pure val PLACEMENTS = PLACEABLE.setOfMaps(GROUPS)

  pure val MAX_SUBMISSIONS = 2
  pure val MAX_LOG = 6

  // --- State -------------------------------------------------------------
  var g: Graph
  var guarded: bool
  var owner: str -> str        // the topology
  var log: List[Event]
  var substrate: View          // shared read models, each folded by its owner
  var cursor: str -> int       // per group, durable
  var hwm: str -> int          // high-water mark of cursor (what is really folded) — a ghost
  var lane: str -> List[Command]
  var pending: str -> List[Command]   // at most one: the claimed, not-yet-run command
  var ran: Set[Command]
  var submitted: int
  var rewound: bool            // ghost: a cursor was lost at some point

  action init = {
    nondet reads = READS.oneOf()
    nondet emits = EMITS.oneOf()
    nondet folds = FOLDS.oneOf()
    nondet reacts = REACTS.oneOf()
    nondet dispatches = DISPATCHES.oneOf()
    nondet interactive = INTERACTIVE.oneOf()
    nondet gd = Set(true, false).oneOf()
    nondet placement = PLACEMENTS.oneOf()
    all {
      g' = { reads: reads, emits: emits, folds: folds, reacts: reacts, dispatches: dispatches, interactive: interactive },
      guarded' = gd,
      owner' = placement,
      log' = [], substrate' = emptyView,
      cursor' = GROUPS.mapBy(h => 0), hwm' = GROUPS.mapBy(h => 0),
      lane' = GROUPS.mapBy(h => []), pending' = GROUPS.mapBy(h => []),
      ran' = Set(), submitted' = 0, rewound' = false,
    }
  }

  // --- Helpers ---------------------------------------------------------------
  pure def claimIndex(gr: Graph, q: List[Command]): int = {
    val urgent = q.indices().filter(i => gr.interactive.contains(q[i].proc))
    if (urgent != Set()) urgent.fold(-1, (a, i) => if (a == -1 or i < a) i else a)
    else 0
  }
  pure def without(q: List[Command], i: int): List[Command] =
    q.slice(0, i).concat(q.slice(i + 1, q.length()))
  pure def route(o: str -> str, lanes: str -> List[Command], cmds: List[Command]): str -> List[Command] =
    cmds.foldl(lanes, (ls, c) => ls.set(o.get(c.proc), ls.get(o.get(c.proc)).append(c)))

  // A group with nothing that consumes the log holds no cursor (topology.md
  // "a procedure group has no cursor"), so it is not gated on folding.
  def consumesLog(h: str): bool =
    MODELS.exists(m => owner.get(m) == h) or POLICIES.exists(y => owner.get(y) == h)

  action fixedConfig = all {
    g' = g, guarded' = guarded, owner' = owner, submitted' = submitted, rewound' = rewound,
  }

  // --- Host submits a command ------------------------------------------------
  action submit = {
    nondet p = PROCS.oneOf()
    val c = { id: { origin: "host", at: submitted, target: p }, proc: p, view: emptyView }
    all {
      submitted < MAX_SUBMISSIONS,
      lane' = lane.set(owner.get(p), lane.get(owner.get(p)).append(c)),
      submitted' = submitted + 1,
      g' = g, guarded' = guarded, owner' = owner, log' = log, substrate' = substrate,
      cursor' = cursor, hwm' = hwm, pending' = pending, ran' = ran, rewound' = rewound,
    }
  }

  // --- Event intake: group h folds ONE entry (Engine.apply_event, h's subset)
  // fold h's projections → commit → run h's policies on the entry → dispatch → save cursor.
  action foldOne = {
    nondet h = GROUPS.oneOf()
    val pos = cursor.get(h)
    all {
      pos < log.length(),
      val e = log[pos]
      val sub2 = MODELS.mapBy(m =>
        if (owner.get(m) == h and g.folds.get(m).contains(e.kind)) substrate.get(m).union(Set(pos))
        else substrate.get(m))
      val dispatched = runPolicies(g, e, pos, sub2).select(c => owner.get(c.id.origin) == h)
      all {
        substrate' = sub2,
        lane' = route(owner, lane, dispatched),
        cursor' = cursor.set(h, pos + 1),
        hwm' = hwm.set(h, if (pos + 1 > hwm.get(h)) pos + 1 else hwm.get(h)),
      },
      fixedConfig, log' = log, pending' = pending, ran' = ran,
    }
  }

  // --- Command intake, step 1: claim (the middle fold has returned) ----------
  action claim = {
    nondet h = GROUPS.oneOf()
    val q = lane.get(h)
    all {
      q.length() > 0,
      pending.get(h).length() == 0,
      not(consumesLog(h)) or cursor.get(h) == log.length(),
      val i = claimIndex(g, q)
      all {
        pending' = pending.set(h, [q[i]]),
        lane' = lane.set(h, without(q, i)),
      },
      fixedConfig, log' = log, substrate' = substrate, cursor' = cursor, hwm' = hwm, ran' = ran,
    }
  }

  // --- Command intake, step 2: run the claimed command (no cursor guard) -------
  action execute = {
    nondet h = GROUPS.oneOf()
    all {
      pending.get(h).length() == 1,
      log.length() < MAX_LOG,
      val cmd = pending.get(h)[0]
      if (guarded and seesOwnBlock(g, log, substrate, cmd))
        all { log' = log, ran' = ran }
      else all {
        log' = log.concat(runProcedure(g, cmd, substrate, log.length())),   // reads the SHARED substrate, now
        ran' = ran.union(Set(cmd)),
      },
      pending' = pending.set(h, []),
      fixedConfig, lane' = lane, substrate' = substrate, cursor' = cursor, hwm' = hwm,
    }
  }

  // --- Fault: at-least-once redelivery of a claimed row ----------------------
  action redeliver = {
    nondet c = ran.oneOf()
    val h = owner.get(c.proc)
    all {
      ran != Set(),
      not(lane.get(h).indices().exists(i => lane.get(h)[i].id == c.id)),
      not(pending.get(h).indices().exists(i => pending.get(h)[i].id == c.id)),
      lane' = lane.set(h, [c].concat(lane.get(h))),
      fixedConfig, log' = log, substrate' = substrate, cursor' = cursor, hwm' = hwm, pending' = pending, ran' = ran,
    }
  }

  // --- Fault: cursor loss (commit-then-save: models keep, position rewinds) ---
  action rewind = {
    nondet h = GROUPS.oneOf()
    nondet back = 0.to(MAX_LOG).oneOf()
    all {
      back < cursor.get(h),
      cursor' = cursor.set(h, back),
      g' = g, guarded' = guarded, owner' = owner, submitted' = submitted, rewound' = true,
      log' = log, substrate' = substrate, hwm' = hwm, lane' = lane, pending' = pending, ran' = ran,
    }
  }

  val allIdle = GROUPS.forall(h => lane.get(h).length() == 0 and pending.get(h).length() == 0 and cursor.get(h) == log.length())
  action quiescent = all {
    submitted == MAX_SUBMISSIONS,
    allIdle or log.length() >= MAX_LOG,
    fixedConfig, log' = log, substrate' = substrate, cursor' = cursor, hwm' = hwm, lane' = lane, pending' = pending, ran' = ran,
  }

  action step = any { submit, foldOne, claim, execute, redeliver, rewind, quiescent }

  // --- Placement predicates (topology.md) ------------------------------------
  // Rule 6: a reader shares a group with every model it reads.
  val rule6 = ELEMS.forall(e => g.reads.get(e).forall(m => owner.get(m) == owner.get(e)))
  // The degenerate topology: everything in one group (st as a topology).
  val oneGroup = GROUPS.exists(h => PLACEABLE.forall(x => owner.get(x) == h))

  // --- Invariants for EVERY topology --------------------------------------------

  // Each model is exactly its owner's fold up to what the owner has really folded.
  val substrateIsFold =
    MODELS.forall(m => substrate.get(m) == foldPrefix(g, log, hwm.get(owner.get(m))).get(m))
  val cursorBelowHwm = GROUPS.forall(h => cursor.get(h) <= hwm.get(h))

  val blocksWellFormed = blocksWellFormedOf(g, log)
  // Exactly-once needs BOTH the application guard and rule 6: a guard sees its
  // own block only through a model its own group folds before it claims.
  val effectsOnce = (guarded and rule6) implies effectsOnceGuardableOf(g, log)
  val causal = causalOf(g, log)

  // A command only ever sits in the lane / pending slot of its procedure's owner.
  val routed = GROUPS.forall(h =>
    lane.get(h).indices().forall(i => owner.get(lane.get(h)[i].proc) == h)
    and pending.get(h).indices().forall(i => owner.get(pending.get(h)[i].proc) == h))

  // Completeness: every reacting policy of every entry its owner has folded
  // has dispatched (queued, claimed, or run); every run left its block.
  val knownIds =
    GROUPS.map(h => lane.get(h).indices().map(i => lane.get(h)[i].id).union(pending.get(h).indices().map(i => pending.get(h)[i].id))).flatten()
      .union(ran.map(c => c.id))
  val dispatchComplete =
    log.indices().forall(pos =>
      POLICIES.forall(y => (g.reacts.get(y).contains(log[pos].kind) and pos < hwm.get(owner.get(y))) implies
        g.dispatches.get(y).forall(p => knownIds.contains({ origin: y, at: pos, target: p }))))
  val emitComplete =
    ran.forall(c => g.emits.get(c.proc) == Set() or log.indices().exists(i => log[i].cause == c.id))

  // --- The theorems ----------------------------------------------------------------
  val freshReads          = freshReadsOf(g, log)
  val freshDispatch       = freshDispatchOf(g, log)
  val causeFolded         = causeFoldedOf(g, log)
  val dispatchCauseFolded = dispatchCauseFoldedOf(g, log)

  // (T1) What the worker promises, made conditional on rule 6: the model had
  //      committed through the command's cause before the run, and no policy
  //      dispatched on a view behind its trigger. NOT freshDispatch: after a
  //      cursor rewind a policy legitimately re-runs on a model that has
  //      committed further (cursors.py commit-then-save), and its re-dispatch
  //      only reaches the log when the application does not, or cannot, guard.
  val rule6Guarantee = rule6 implies (causeFolded and dispatchCauseFolded)

  // (T1') In the no-fault world the strong form does hold.
  val noFaultFreshDispatch = (rule6 and not(rewound)) implies freshDispatch

  // (T2) st falls out as a topology: one group ⇒ st's fresh reads.
  val singleGroupIsSt = oneGroup implies (freshReads and dispatchCauseFolded and (not(rewound) implies freshDispatch))

  val allInvariants = all {
    substrateIsFold, cursorBelowHwm, blocksWellFormed, effectsOnce, causal, routed,
    dispatchComplete, emitComplete, rule6Guarantee, noFaultFreshDispatch, singleGroupIsSt
  }

  // --- Witnesses ---------------------------------------------------------------------
  // The claim→execute window: a rule-6-LEGAL split reads behind the tail.
  // (May still be serializable — a block reorder between claim and execute.)
  val witnessClaimWindow  = rule6 and not(freshReads)
  // Write skew under rule 6: two decisions, neither saw the other, no serial
  // order explains both. THE counter-example to "byte-identical models".
  val witnessWriteSkew    = rule6 and writeSkewOf(g, log)
  // A rule-6-VIOLATING split breaks even the weak guarantee.
  val witnessStaleRead    = not(rule6) and not(causeFolded)
  val witnessStalePolicy  = not(rule6) and not(freshDispatch)
  val witnessLegalLag     = rule6 and not(oneGroup) and GROUPS.exists(h => cursor.get(h) < log.length()) and log.length() >= 2
  val witnessCrossGroup   = log.indices().exists(i => log[i].cause.origin != "host" and owner.get(log[i].cause.origin) != owner.get(log[i].cause.target))
  val witnessRedelivered  = GROUPS.exists(h => lane.get(h).indices().exists(i => ran.exists(c => c.id == lane.get(h)[i].id)))
  val witnessRewound      = GROUPS.exists(h => cursor.get(h) < hwm.get(h))
  val witnessRedispatch   = witnessRewound and GROUPS.exists(h => lane.get(h).indices().exists(i => ran.exists(c => c.id == lane.get(h)[i].id)))
  val witnessDoubleAppend = not(effectsOnceOf(log))
  val witnessUnguardable  = guarded and rule6 and not(effectsOnceOf(log))
  // A guarding, guardable procedure still double-appends when rule 6 is broken.
  val witnessGuardMissed  = guarded and not(rule6) and not(effectsOnceGuardableOf(g, log))
  val witnessOneGroupRun  = oneGroup and log.length() >= 2
}
